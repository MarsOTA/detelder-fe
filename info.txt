//Per copiare assets su dist
npm install vite-plugin-static-copy --save-dev
+ sul file vite.config.js
viteStaticCopy({
      targets: [
        {
          src: 'src/assets', // sorgente da copiare
          dest: ''            // destinazione dentro /dist
        }
      ]
})

Per la PWA (installazione minimal senza icone)
npm install vite-plugin-pwa --save-dev
+ sul file vite.config.js
VitePWA({
      registerType: 'autoUpdate',
      manifest: {
        name: 'La mia app React',
        short_name: 'ReactApp',
        start_url: '/',
        display: 'standalone',
        background_color: '#ffffff',
        theme_color: '#ffffff',
        // 🔹 Niente "icons": il sistema userà un’icona di default
      },
      }),
